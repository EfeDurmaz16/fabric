//! The check of off_runtime.rs, written against the old client, which
//! took the journal itself: where does each log call run?
#![allow(clippy::unwrap_used, reason = "test")]
mod common;
use std::sync::{Arc, Mutex};
use common::{create, journal_for, scope};
use fabric_bridge::{Bridge, Config, Entry, LogError, OperationLog};
use fabric_core::managed::OperationId;
use fabric_journal::{Journal, ProviderAnswer, ProviderDispatch, ProviderStatus};
type Seen = Arc<Mutex<Vec<(&'static str, String)>>>;
struct Watched { journal: Journal, seen: Seen }
impl Watched {
    fn note(&self, call: &'static str) {
        let thread = std::thread::current().name().unwrap_or("").to_owned();
        self.seen.lock().unwrap().push((call, thread));
    }
}
impl OperationLog for Watched {
    fn begin(&mut self, d: &ProviderDispatch, r: Option<&ProviderAnswer>) -> Result<(), LogError> { self.note("begin"); self.journal.begin(d, r) }
    fn record(&mut self, o: OperationId, s: ProviderStatus, a: &ProviderAnswer, l: Option<&str>) -> Result<(), LogError> { self.note("record"); self.journal.record(o, s, a, l) }
    fn locate(&mut self, o: OperationId, l: &str) -> Result<(), LogError> { self.note("locate"); self.journal.locate(o, l) }
    fn entry(&self, o: OperationId) -> Result<Option<Entry>, LogError> { self.note("entry"); self.journal.entry(o) }
}
#[tokio::test]
async fn every_log_call_runs_on_the_log_thread() {
    let dir = tempfile::tempdir().unwrap();
    let seen = Seen::default();
    let mut log = Watched { journal: journal_for(dir.path(), scope()), seen: Arc::clone(&seen) };
    let mut config = Config::new("fixture");
    config.handshake_timeout = std::time::Duration::from_secs(5);
    let bridge = Bridge::spawn(tokio::process::Command::new(env!("CARGO_BIN_EXE_fabric-bridge-double")), config).await.unwrap();
    let done = bridge.request(&mut log, create(1, "fixture/ok")).await.unwrap().finish(&mut log).await.unwrap();
    assert_eq!(done.status, ProviderStatus::Succeeded);
    for (call, thread) in seen.lock().unwrap().iter() {
        assert_eq!(thread, "fabric-bridge-log", "{call} ran on {thread:?}");
    }
}
